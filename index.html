# Script a executer dans QGIS pour generer les PNG web-compatibles # + fichier de donnees JSON pour l'affichage au survol de la souris import numpy as np from osgeo import gdal import json import matplotlib matplotlib.use("Agg") import matplotlib.pyplot as plt import matplotlib.colors as mcolors import os OUTPUT_WEB_DIR = r"C:\Users\cedfa\webmap" os.makedirs(OUTPUT_WEB_DIR, exist_ok=True) def tif_to_png_and_data(tif_path, png_path, data_path, vmin, vmax, cmap_name="RdYlBu_r", decim=2): ds = gdal.Open(tif_path) gt = ds.GetGeoTransform() data = ds.GetRasterBand(1).ReadAsArray().astype(float) xs, ys = ds.RasterXSize, ds.RasterYSize lon_min = gt[0] lon_max = gt[0] + xs * gt[1] lat_max = gt[3] lat_min = gt[3] + ys * gt[5] cmap = plt.get_cmap(cmap_name) norm = mcolors.Normalize(vmin=vmin, vmax=vmax) rgba = cmap(norm(data)) rgba[np.isnan(data)] = [0, 0, 0, 0] plt.imsave(png_path, rgba, origin="upper") # Decimation pour reduire la taille du fichier de donnees # (on garde 1 pixel sur "decim" pour alleger le JSON) data_decim = data[::decim, ::decim] nrows_d, ncols_d = data_decim.shape # Remplacer les NaN par null pour le JSON data_list = [] for row in data_decim: data_list.append([None if np.isnan(v) else round(float(v), 1) for v in row]) data_export = { "lat_min": round(lat_min, 5), "lat_max": round(lat_max, 5), "lon_min": round(lon_min, 5), "lon_max": round(lon_max, 5), "nrows": nrows_d, "ncols": ncols_d, "values": data_list, } with open(data_path, "w", encoding="utf-8") as f: json.dump(data_export, f) return { "lat_min": round(lat_min, 5), "lat_max": round(lat_max, 5), "lon_min": round(lon_min, 5), "lon_max": round(lon_max, 5), } import datetime as dt today_str = dt.date.today().strftime("%Y%m%d") LAYERS = [ { "id": "tn_vercors", "nom": "Tn Vercors", "tif": f"C:/Users/cedfa/cartes_tn/tn_vercors_{today_str}.tif", "vmin": -5, "vmax": 20, "decim": 1, }, { "id": "tn_bagnols", "nom": "Tn Bagnols", "tif": f"C:/Users/cedfa/cartes_tn_bagnols/tn_bagnols_{today_str}.tif", "vmin": 5, "vmax": 25, "decim": 2, }, { "id": "tx_bagnols", "nom": "Tx Bagnols", "tif": f"C:/Users/cedfa/cartes_tx_bagnols/tx_bagnols_{today_str}.tif", "vmin": 20, "vmax": 40, "decim": 2, }, ] bounds_all = {} for layer in LAYERS: if os.path.exists(layer["tif"]): png_path = os.path.join(OUTPUT_WEB_DIR, layer["id"] + ".png") data_path = os.path.join(OUTPUT_WEB_DIR, layer["id"] + "_data.json") bounds = tif_to_png_and_data(layer["tif"], png_path, data_path, layer["vmin"], layer["vmax"], decim=layer["decim"]) bounds_all[layer["id"]] = {**bounds, "nom": layer["nom"], "date": today_str} print(f"OK: {layer['nom']} -> {png_path} + {data_path}") else: print(f"MANQUANT: {layer['tif']}") with open(os.path.join(OUTPUT_WEB_DIR, "bounds.json"), "w", encoding="utf-8") as f: json.dump(bounds_all, f, indent=2) print("\nFichiers prets dans:", OUTPUT_WEB_DIR) print("A uploader sur GitHub: les 3 PNG, les 3 _data.json, et bounds.json")
